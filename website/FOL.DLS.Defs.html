Section Retracts.

  Class retract (A B: Type): Type := {
    retr_i: A → B;
    retr_s: B → A;
    retr_o: ∀ x, retr_s (retr_i x) = x;
  }.

  Class strongInf X := sinf:> (retract (list X) X).

  Class inhab (X: Type) := elem: X.

End Retracts.
Notation "A ≤R B" := (retract A B) (at level 80).

Section Properties.

  Definition totalR {X Y: Type} (R: X → Y → Prop) :=
    ∀ x, ∃ y, R x y.

  Definition directedR {X: Type} (R: X → X → Prop) :=
    ∀ x x', ∃ y, R x y ∧ R x' y.

End Properties.
Section LogicalPrinciples.

  Definition LEM :=
    ∀ P, P ∨ ¬ P.

  Definition EP_on X :=
    ∀ (P: X → Prop),
    ∃ w: X,
    (∃ x, P x) → P w.

  Definition DP_on X :=
    ∀ (P: X → Prop),
    ∃ w: X,
    P w → ∀ x, P x.

  Definition BEP_on K X :=
    ∀ (P: X → Prop),
    ∃ f: K → X,
    (∃ x, P x) → (∃ k, P (f k)).

  Definition BDP_on K X :=
    ∀ (P: X → Prop),
    ∃ f: K → X,
    (∀ k, P (f k)) → (∀ x, P x).

  Definition GMP K :=
    ∀ f: K → bool,
    ¬ ¬ (∃ k, f k = true) → ∃ k, f k = true.

  Definition AC_on K Y :=
    ∀ R: K → Y → Prop, totalR R →
    ∃ f: K → Y,
    ∀ k, R k (f k).

  Definition DC_on Y :=
    ∀ R: Y → Y → Prop, totalR R →
    ∃ f: nat → Y,
    ∀ n, R (f n) (f (S n)).

  Definition OAC_on K Y :=
    ∀ R: K → Y → Prop,
    ∃ f: K → Y,
    totalR R → ∀ k, R k (f k).

  Definition BAC_on K Y :=
    ∀ R: K → Y → Prop, totalR R →
    ∃ f: K → Y,
    totalR (fun k0 k ⇒ R k0 (f k)).

  Definition DDC_on K Y :=
    ∀ R: Y → Y → Prop, directedR R →
    ∃ f: K → Y,
    directedR (fun k k' ⇒ R (f k) (f k')).

  Definition OBAC_on K Y :=
    ∀ R: K → Y → Prop,
    ∃ f: K → Y,
    totalR R → totalR (fun k0 k ⇒ R k0 (f k)).

  Definition ODDC_on K Y :=
    ∀ R: Y → Y → Prop,
    ∃ f: K → Y,
    directedR R → directedR (fun k k' ⇒ R (f k) (f k')).

  Definition EP := ∀ X, inhab X → EP_on X.
  Definition DP := ∀ X, inhab X → DP_on X.
  Definition BEP K := ∀ X, inhab X → BEP_on K X.
  Definition BDP K := ∀ X, inhab X → BDP_on K X.
  Definition AC K := ∀ Y, inhab Y → AC_on K Y.
  Definition DC := ∀ Y, inhab Y → DC_on Y.
  Definition OAC K := ∀ Y, inhab Y → OAC_on K Y.
  Definition BAC K := ∀ Y, inhab Y → BAC_on K Y.
  Definition DDC K := ∀ Y, inhab Y → DDC_on K Y.
  Definition OBAC K := ∀ Y, inhab Y → OBAC_on K Y.
  Definition ODDC K := ∀ Y, inhab Y → ODDC_on K Y.

End LogicalPrinciples.
Section DLS.

  Context {fff: falsity_flag}.

  Definition DLS'_on
    {s_f: funcs_signature}
    {s_P: preds_signature}
    (K: Type)
    (M: model) :=
    ∃ (Is: interp K), Build_model Is ⪳ M.

  Definition DLS'
    {s_f: funcs_signature}
    {s_P: preds_signature}
    (K: Type) :=
    ∀ (M: model) (m0: M), DLS'_on K M.
  #[global] Arguments DLS' _ _ _, {_ _} _.

  Definition DLS_on
    {s_f: funcs_signature}
    {s_P: preds_signature}
    (K: Type)
    (M: model) :=
    ∃ (N: model), N ⪳ M ∧ inhabited (N ≤R K).

  Definition DLS
    {s_f: funcs_signature}
    {s_P: preds_signature}
    (K: Type) :=
    ∀ (M: model), M → DLS_on K M.
   #[global] Arguments DLS _ _ _, {_ _} _.

End DLS.

Section MainStatement.

  Definition DLS_iff_BEP_BDP_BAC_DDC := ∀ K, inhabited (strongInf K) →
  ((BDP K ∧ BEP K) ∧ (DDC K ∧ BAC K)) ↔
  (∀ (s_f: funcs_signature) (s_P: preds_signature), inhab s_P →
    s_f + s_P ≤R K → DLS K).

End MainStatement.
