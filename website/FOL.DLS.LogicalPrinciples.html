  Fact prv_BAC_K_K: ∀ K, inhab K → BAC_on K K.
  Proof.
    intros K k0 R htR.
    ∃ id.
    exact htR.
  Qed.

    ∃ id.

  Lemma BAC_of_AC:
  ∀ K, inhab K →
  ∀ Y, inhab Y →
  AC_on K Y → BAC_on K Y.
  Proof.
    intros K k0 Y y0 ac R htR.
    destruct (ac _ htR) as [f hf].
    ∃ f.
    intros k. ∃ k. apply hf.
  Qed.

  Lemma AC_K_Y_iff_AC_K_K_and_BAC_K_Y:
  ∀ K, inhab K →
  ∀ Y, inhab Y →
  (BAC_on K Y ∧ AC_on K K) → AC_on K Y.
  Proof.
    intros K k0 Y y0 [bac ac] R htR.
    destruct (bac R htR) as [f hf].
    destruct (ac (fun k k' ⇒ R k (f k')) hf) as [g hg].
    ∃ (fun k ⇒ f (g k)).
    exact hg.
  Qed.

  Lemma AC_K_iff_BAC_K_of_AC_K_K:
  ∀ K, inhab K →
  AC_on K K →
  (BAC K ↔ AC K).
  Proof.
    intros K k0 acKK. split.
    - intros bac Y y0. refine (AC_K_Y_iff_AC_K_K_and_BAC_K_Y _ _ _ _ (conj _ _)).
      + apply (bac _ _).
      + apply acKK.
    - intros ac Y y0. refine (BAC_of_AC _ _ _ _ (ac _ _)).
  Qed.

  Lemma AC_K_iff_AC_K_K_and_BAC_K:
  ∀ K, inhab K →
  (AC K ↔
  (BAC K ∧ AC_on K K)).
  Proof.
    intros K k0. split.
    - intros ac. split.
      + intros Y y0 R htR.
        destruct (ac _ _ R htR) as [f hf].
        ∃ f.
        intros k. ∃ k. apply hf.
      + apply (ac _ _).
    - intros [bac ac] Y y0 R htR.
      destruct (bac _ _ R htR) as [f hf].
      destruct (ac (fun k k' ⇒ R k (f k')) hf) as [g hg].
      ∃ (fun k ⇒ f (g k)).
      exact hg.
  Qed.

  Lemma DC_of_DDC_K_and_AC_K {K} {siK: strongInf K}:
  (DDC K ∧ AC K) → DC.
  Proof.
    intros [ddc ac] Y y0 R htR.
    pose proof (z := _ : K × K ≤R K).
    pose proof (p := _ : K + K ≤R K).
    pose (Q := fun (r r': K → Y) ⇒ (∀ n, ∃ m, r n = r' m) ∧
      ∀ n, ∃ k, R (r n) (r' k)).
    assert (htQ: totalR Q).
    {
      intros r.
      pose (T := fun n x ⇒ R (r n) x).
      assert (htT: totalR T).
      {
        intros n. unfold T.
        destruct (htR (r n)) as [y hy].
        ∃ y. apply hy.
      }
      unfold Q.
      destruct (ac _ _ T htT) as [s hs].
      ∃ (fun n ⇒ match retr_s p n with
        | inl n' ⇒ r n'
        | inr n' ⇒ s n' end).
      split.
      + intros n.
        ∃ (retr_i p (inl n)). rewrite retr_o. reflexivity.
      + intros n. unfold T in hs.
        ∃ (retr_i p (inr n)). rewrite retr_o.
        apply hs.
    }
    assert (hdQ: directedR Q).
    {
      intros r r'.
      destruct (htQ r) as [s [h1s h2s]].
      destruct (htQ r') as [s' [h1s' h2s']].
      ∃ (fun n ⇒ match retr_s p n with | inl n' ⇒ s n' | inr n' ⇒ s' n' end).
      repeat split.
      + intros n. destruct (h1s n) as [m hm].
        ∃ (retr_i p (inl m)). rewrite retr_o. apply hm.
      + intros n. destruct (h2s n) as [m hm].
        ∃ (retr_i p (inl m)). rewrite retr_o. apply hm.
      + intros n. destruct (h1s' n) as [m hm].
        ∃ (retr_i p (inr m)). rewrite retr_o. apply hm.
      + intros n. destruct (h2s' n) as [m hm].
        ∃ (retr_i p (inr m)). rewrite retr_o. apply hm.
    }
    destruct (ddc (K → Y) (fun _ ⇒ y0) Q hdQ) as [F hF].
    pose (rho := fun n ⇒ let (m, k) := retr_s z n in F m k).
    assert (hrho: Q rho rho).
    {
      split.
      + intros n. ∃ n. reflexivity.
      + intros n. unfold Q in hF. unfold rho.
        destruct (retr_s z n) as [n1 n2] eqn: en.
        destruct (hF n1 n2) as [m1 [[_ hm1] _]].
        destruct (hm1 n2) as [m2 hm2].
        ∃ (retr_i z (m1, m2)). rewrite retr_o. apply hm2.
    }
    destruct (hrho) as [_ H'].
    pose (U := fun n n' ⇒ R (rho n) (rho n')).
    destruct (ac _ _ U H') as [g hg].
    pose (G := fix G n k0 := match n with | 0 ⇒ k0 | S n' ⇒ g (G n' k0) end).
    ∃ (fun n ⇒ rho (G n (_: inhab K))).
    intros n. apply (hg).
  Qed.

  Lemma DDC_mono {K K'} {r: K ≤R K'}:
  DDC K → DDC K'.
  Proof.
    intros ddc Y y0 R hdR.
    destruct (ddc _ _ _ hdR) as [f hf].
    ∃ (fun k' ⇒ f (retr_s r k')).
    intros k1' k2'.
    destruct (hf (retr_s r k1') (retr_s r k2')) as [k0 hk0].
    ∃ (retr_i r k0).
    rewrite retr_o.
    exact hk0.
  Qed.

End BelowAC.

  Lemma BEP_mono {K K'} {r: K ≤R K'}:
  ∀ X, inhab X →
  BEP_on K X → BEP_on K' X.
  Proof.
    intros X _ bep P.
    destruct (bep P) as [w hw].
    ∃ (fun k' ⇒ w (retr_s r k')).
    intros H; destruct (hw H) as [k hk]; ∃ (retr_i r k); rewrite retr_o; apply hk.
  Qed.

  Lemma BDP_mono {K K'} {r: K ≤R K'}:
  ∀ X, inhab X →
  BDP_on K X → BDP_on K' X.
  Proof.
    intros X _ bdp P.
    destruct (bdp P) as [w hw].
    ∃ (fun k' ⇒ w (retr_s r k')).
    intros H; apply hw; intros k; specialize (H (retr_i r k)); rewrite retr_o in H; apply H.
  Qed.

  Lemma EP_iff_BEP_unit:
  EP ↔ BEP unit.
  Proof.
    split.
    all: intros lp X x0 P.
    all: destruct (lp _ _ P) as [w hw].
    - ∃ (fun _ ⇒ w); intros H. ∃ tt; apply hw, H.
    - ∃ (w tt); intros H. destruct (hw H) as [[] hw']; apply hw'.
  Qed.

  Lemma DP_iff_BDP_unit:
  DP ↔ BDP unit.
  Proof.
    split.
    all: intros lp X x0 P.
    all: destruct (lp _ _ P) as [w hw].
    - ∃ (fun _ ⇒ w); intros H. apply hw, H, tt.
    - ∃ (w tt); intros H. apply hw; intros []; apply H.
  Qed.

  Lemma LEM_of_BEP_and_MP: ∀ B, BEP B ∧ GMP B → LEM.
  Proof.
    intros B [bep mp] p.
    pose (A := {b: bool | b = false ∨ (p ∨ ¬ p)}).
    pose (P := fun (a: A) ⇒ let (b, _) := a in match b with | true ⇒ p ∨ ¬ p | false ⇒ False end).
    destruct (bep A (exist _ false (or_introl (eq_refl))) P) as [f hf].
    pose (f' := fun b ⇒ let (a, _) := f b in a).
    assert (H: p ∨ ¬ p ↔ (∃ c, P c)).
    {
      split.
      + intros e.
        × ∃ (exist _ true (or_intror e)). unfold P. exact e.
      + intros [[[|] hb] ha]. all: unfold P in ha.
      2: exfalso.
      all: exact ha.
    }
    assert (H': (∃ b, f' b = true) ↔ (∃ a, P a)).
    {
      split.
      + intros [b hb]. ∃ (f b). unfold P. unfold f' in hb. destruct (f b) as [x [abs|e]].
        × rewrite hb in abs. inversion abs.
        × rewrite hb. exact e.
      + intros h. destruct (hf h) as [b hb]. unfold P in hb.
        ∃ b. unfold f'. destruct (f b) as [[|] hb'].
        × reflexivity.
        × exfalso. apply hb.
    }
    rewrite H, <-H'. apply (mp f'). rewrite H', <-H.
    firstorder.
  Qed.

  Lemma prv_MP_option {K} `{k0: inhab K}: GMP K → GMP (option K).
  Proof.
    intros mp f.
    destruct (f None) as [|] eqn: eN.
    - intros _. ∃ None. exact eN.
    - intros H.
      destruct (mp (fun k ⇒ f (Some k))) as [k hk].
      + intros abs. apply H. intros [[k|] hk].
        × apply abs. ∃ k; apply hk.
        × rewrite eN in hk; discriminate hk.
      + ∃ (Some k). exact hk.
  Qed.

  Lemma DDC_of_ODDC {K}:
  ODDC K → DDC K.
  Proof.
    intros oddc Y y0 R hdR.
    destruct (oddc _ _ R) as [f hf].
    ∃ f.
    apply (hf hdR).
  Qed.

  Lemma BAC_of_OBAC {K}:
  OBAC K → BAC K.
  Proof.
    intros obac Y y0 R htR.
    destruct (obac _ _ R) as [f hf].
    ∃ f.
    apply (hf htR).
  Qed.

  Fact OBAC_iff_BAC_and_BEP: ∀ K, strongInf K →
  (OBAC K) ↔
  ((BAC K) ∧
  (BEP K)).
  Proof.
    intros K siK. split.
    - intros obac.
      split.
      + intros Y y0 R htR.
        destruct (obac Y y0 R) as [f hf].
        ∃ f.
        apply (hf htR).
      + intros X x0 P.
        destruct (obac X _ (fun _ x ⇒ P x)) as [w hw].
        ∃ (w).
        intros [x hx].
        refine (hw _ (_: inhab K)).
        intros _. ∃ x; exact hx.
    - intros [bac bep] Y y0 R.
      pose proof (z:= _ : K × K ≤R K).
      destruct (bac _ y0 (fun _ _ ⇒ True) (fun _ ⇒ ex_intro _ y0 I)) as [dummy _].
      pose (P := fun f: K → Y ⇒ totalR (fun k k' ⇒ R k (f k'))).
      destruct (bep _ dummy P) as [f hf].
      ∃ (fun k ⇒ let (k1, k2) := retr_s z k in f k1 k2).
      intros htR k.
      destruct (bac Y y0 R htR) as [g hg].
      assert (∃ g, P g).
      { ∃ g. intros k'. apply (hg k'). }
      destruct (hf H) as [k1 hk1].
      destruct (hk1 k) as [k2 hk2].
      ∃ (retr_i z (k1, k2)). rewrite retr_o.
      apply hk2.
  Qed.

  Fact AC_and_EP_of_OAC:
  ∀ K, inhab K →
  OAC K → (AC K ∧ EP).
  Proof.
    intros K k0 oac.
    split.
    + intros Y y0 R htR.
      destruct (oac _ _ R) as [f hf].
      ∃ f.
      apply (hf htR).
    + intros X x0 P. unfold OAC, OAC_on in oac.
      destruct (oac X _ (fun _ z ⇒ P z)) as [w hw].
      ∃ (w k0).
      intros [x hx].
      apply hw.
      intros _.
      ∃ x. apply hx.
  Qed.

End OmniscientChoiceAxiom.
